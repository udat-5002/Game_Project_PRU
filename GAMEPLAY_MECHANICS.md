### Detection Mechanics

#### Light Visibility
```csharp
public class DetectionSystem
{
    // Brightness level affects detection
    int GetVisibilityLevel()
    {
        // 0 = completely dark
        // 100 = bright daylight
    }
    
    // Guard can see player if:
    bool CanSeePlayer()
    {
        float distanceToPlayer = Vector3.Distance(guardPos, playerPos);
        float visionRange = 20f + (visibilityLevel * 0.2f);
        bool inLineOfSight = Physics.Raycast(guardEyes, playerPos);
        
        return distanceToPlayer < visionRange && inLineOfSight;
    }
}
```

```csharp
public enum GuardState
```

### Alarm System
```csharp
public class AlarmSystem
{
    // When guard goes COMBAT state:
    void TriggerAlarm()
    {
        alarmSoundEffect.Play();
        
        // All guards nearby get alert (radius 50m)
        BroadcastAlert();
        
        // Reinforcements called
        SpawnGuardReinforcements(reinforcementDelay: 15f);
        
        // Mission fails if player captured
    }
    
    // Player must either:
    // 1. Hide successfully
    // 2. Escape alarm radius
    // 3. Reach objective and escape
}
```

### Stealth Failure
```
IF guard catches player:
  - Cutscene: Guard capturing
  - Text: "You were captured..."
  - Load checkpoint (no damage, just restart section)
  - No death counter (non-punitive)
```

#### Delivery Quest
```csharp
public class DeliveryQuest : Quest
{
    public Letter[] lettersToDeliver;
    public int lettersDelivered = 0;
    
    public void DeliverLetter(Letter letter, NPC recipient)
    {
        if (letter.recipientID == recipient.npcID)
        {
            lettersDelivered++;
            recipient.OnLetterReceived(letter);
            
            if (lettersDelivered >= lettersToDeliver.Length)
            {
                CompleteQuest();
            }
        }
    }
}
```

#### Exploration Quest
```csharp
public class ExplorationQuest : Quest
{
    public Item[] targetItems;
    public Vector3[] searchAreas;
    
    void SearchArea(Vector3 position)
    {
        // Find hidden items within radius
        foreach(Item item in FindItemsNearby(position))
        {
            if (IsTargetItem(item))
            {
                item.Highlight();
            }
        }
    }
}
```

#### Survival Quest
```csharp
public class SurvivalQuest : Quest
{
    public Vector3 targetLocation;
    public float timeLimit = 300f;  // seconds
    public bool allowDetection = false;
    
    void Update()
    {
        if (isActive)
        {
            timeLimit -= Time.deltaTime;
            
            if (timeLimit <= 0)
            {
                FailQuest("Time expired");
            }
            
            if (!allowDetection && IsPlayerDetected())
            {
                FailQuest("You were discovered");
            }
        }
    }
}
```
